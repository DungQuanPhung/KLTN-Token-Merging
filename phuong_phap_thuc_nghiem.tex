\subsection{3.1.2. Giải thích từng stage}

\paragraph{Stage 1 — Tiền xử lý.}
Mỗi cặp (câu, aspect term) được token hóa theo định dạng \textbf{Sentence-Pair
Classification (SPC)}:
\[\texttt{[CLS]}\ \underbrace{\text{câu}}_{\text{segment A}}\ \texttt{[SEP]}\
\underbrace{\text{aspect term}}_{\text{segment B}}\ \texttt{[SEP]},\quad L_{\max}=128.\]
Đồng thời, \textbf{vector chỉ thị khía cạnh} \texttt{lcf\_vec}
$\in\{0,1\}^{128}$ được xây dựng bằng cách so sánh \texttt{offset\_mapping}
của tokenizer với khoảng ký tự của cụm khía cạnh trong segment A.
Nếu cụm khía cạnh không xuất hiện trong segment A (cắt quá độ dài),
hệ thống dùng \texttt{token\_type\_ids} của segment B làm chỉ thị dự phòng.

\paragraph{Stage 2 — ATE bằng T5.}
\texttt{T5AspectExtractor} nhận câu và sinh danh sách khía cạnh theo
định dạng GAS: \texttt{"(term1); (term2)"} hoặc \texttt{"none"}.
Beam search với $\text{beam}=4$, \texttt{max\_len}=64. Đầu ra được
\textbf{chuẩn hóa Levenshtein} để ánh xạ về chuỗi con thực sự có trong
câu, tránh lỗi chính tả hay subword không tồn tại.

\paragraph{Stage 3 — APC đa nhiệm.}
Với mỗi cặp (câu, aspect term), \texttt{FastLcfBertMultiTask} dự đoán
đồng thời category và sentiment. Đây là stage trung tâm -- chi tiết ở
Mục~\ref{subsec:apc_arch}.

\paragraph{Stage 4 — Tổng hợp.}
Aspect term từ Stage 2 (hoặc gold) ghép với category và sentiment từ
Stage 3 thành bộ ba $(a_i,c_i,s_i)$.

\section{3.2. Các phương pháp}
\label{sec:methods}

\subsection{3.2.1. Phương pháp 1: Local Context Focus (LCF)}
\label{subsec:pp_lcf}

\paragraph{Xây dựng lcf\_vec (Stage 1).}
Trong hàm \texttt{build\_lcf\_vec()} (\texttt{dataset\_utils.py}),
hệ thống dùng \texttt{offset\_mapping} của tokenizer để xác định tập
chỉ số $\mathcal{A}=\{i:[\ell_i,r_i]\cap[a_s,a_e]\neq\emptyset\}$
(các sub-word token trong segment A giao với khoảng ký tự của cụm khía
cạnh), rồi đặt $\texttt{lcf\_vec}[i]=1$ với $i\in\mathcal{A}$.

\paragraph{Tính SRD.}
Từ $\mathcal{A}$, xác định $a_{start}=\min\mathcal{A}$,
$a_{end}=\max\mathcal{A}$. SRD tại vị trí $i$:
\begin{equation}
SRD_i = \max\!\Big(0,\,|i-\text{center}|-\text{half}\Big),\quad
\text{center}=\frac{a_s+a_e}{2},\quad
\text{half}=\left\lfloor\frac{a_e-a_s+1}{2}\right\rfloor
\label{eq:srd_method}
\end{equation}
Token có $SRD_i\le\alpha=5$ thuộc ngữ cảnh cục bộ.

\paragraph{Hai biến thể trọng số.}
\begin{itemize}
  \item \textbf{CDW} (mặc định, \texttt{use\_cdm=False}):
    $c_i=1$ nếu $SRD_i\le\alpha$; ngược lại $c_i=(L-(SRD_i-\alpha))/L$,
    kẹp $[0,1]$.
  \item \textbf{CDM} (\texttt{use\_cdm=True}):
    $m_i=\mathbf{1}[SRD_i\le\alpha]$.
\end{itemize}

\paragraph{Tích hợp hai luồng.}
\begin{align}
H_{local} &= H \odot c \quad\text{(CDW) hoặc}\quad H \odot m \quad\text{(CDM)} \label{eq:lcf_local}\\
H_{fused} &= \texttt{Linear}_{2d\to d}\big([H_{local}\,;\,H]\big) \label{eq:lcf_fuse}
\end{align}
$H_{fused}$ qua dropout, self-attention \texttt{bert\_SA\_}, pooler
tại \texttt{[CLS]}, rồi vào hai đầu phân loại.

\subsection{3.2.2. Phương pháp 2: Token Merging (ToMe)}
\label{subsec:pp_tome}

Module \texttt{ToMeSequenceMerger} được chèn \textbf{sau backbone,
trước LCF}. Lý do vị trí này:
\begin{enumerate}
  \item Backbone đã tiền huấn luyện -- chèn vào giữa làm thay đổi phân
        bố đầu vào các lớp sau;
  \item Chèn trước LCF cho phép \texttt{lcf\_vec} được cập nhật đồng bộ
        theo chuỗi đã gộp (qua quy tắc $\max$);
  \item Cả hai nhánh Fusion làm việc trên cùng độ dài chuỗi, tránh không
        đồng bộ.
\end{enumerate}

\subsubsection{3.2.2.1. Chiến lược 1 -- Bipartite}

\begin{enumerate}
  \item Xác định pool nội bộ: loại trừ \texttt{[CLS]}, \texttt{[SEP]}
        và (khi \texttt{protect\_aspect=True}) token khía cạnh.
  \item Gán pool xen kẽ vào hai nhánh $A=\{p_0,p_2,p_4,\ldots\}$,
        $B=\{p_1,p_3,p_5,\ldots\}$ theo thứ hạng trong pool.
  \item Chuẩn hóa $L_2$: $\hat{x}_i = x_i/\|x_i\|$.
  \item Mỗi $a\in A$ chọn $b^*\in B$ có cosine lớn nhất (mỗi $b$ tối đa
        một lần -- khử trùng theo thứ tự).
  \item Hợp nhất: $x_a \leftarrow \tfrac12(x_a+x_{b^*})$;
        $\texttt{lcf}[a]\leftarrow\max(\texttt{lcf}[a],\texttt{lcf}[b^*])$;
        vị trí $b^*$ bị xóa.
  \item Lặp \texttt{TOME\_MERGE\_STEPS=2} lần.
\end{enumerate}

\subsubsection{3.2.2.2. Chiến lược 2 -- Sequential Local}

\begin{enumerate}
  \item Phát hiện \textbf{ranh giới LCF (mid\_sep)}: vị trí $j$ là ranh
        giới khi $\texttt{lcf}[j]<0.5$ và $\texttt{lcf}[j+1]>0.5$; các
        vị trí này không bị xóa.
  \item Quét trái$\to$phải qua các token còn hoạt động
        (không thuộc CLS, SEP, mid\_sep, aspect).
  \item Tại token $i$: tính $\text{sim}_{left}$, $\text{sim}_{right}$
        với hàng xóm gần nhất trong vùng hợp lệ.
  \item Nếu $\text{sim}_{left}>\text{sim}_{right}$: $i$ gập vào trái
        ($x_{left}\leftarrow\tfrac12(x_{left}+x_i)$, xóa $i$); ngược lại
        phải gập vào $i$ ($x_i\leftarrow\tfrac12(x_i+x_{right})$, xóa phải).
  \item Một lượt quét = một bước gộp; có thể xảy ra \textbf{hiệu ứng cascade}
        khiến nhiều token cùng gập về một điểm.
\end{enumerate}

\subsubsection{3.2.2.3. Chiến lược 3 -- Attention-Weighted}

\begin{enumerate}
  \item Xếp hạng token theo \texttt{attn\_seg} tăng dần.
  \item Xác định tập bảo vệ: \texttt{[CLS]}, \texttt{[SEP]}, khía cạnh
        ($\texttt{lcf}>0.5$), top 25\% attention.
  \item Lặp đến khi gộp đủ $\lfloor(n-\texttt{prot\_l}-\texttt{prot\_r})/2\rfloor$ lần:
    \begin{enumerate}
      \item Chọn token $i$ có attention thấp nhất chưa bị loại và không được bảo vệ;
      \item Tìm hàng xóm $j$ (không bị loại, không phải khía cạnh) có cosine cao nhất;
      \item $x_j\leftarrow\tfrac12(x_j+x_i)$;
            $\texttt{lcf}[j]\leftarrow\max(\texttt{lcf}[j],\texttt{lcf}[i])$;
            đánh dấu $i$ đã bị loại.
    \end{enumerate}
\end{enumerate}

\subsubsection{3.2.2.4. Chế độ resize}
\label{subsubsec:why_resize}

Sau khi gộp xuống $L'\le L$, chuỗi được nội suy tuyến tính trở lại $L$:
\begin{equation}
H_{\text{out}} = \texttt{F.interpolate}(H', \text{size}=L, \text{mode}=\texttt{"linear"})
\end{equation}
\texttt{lcf\_vec} cũng được nội suy nên vùng khía cạnh ``loang'' thành
dải liên tục -- LCF phía sau vẫn hoạt động đúng.

\noindent\textbf{Lý do chỉ dùng resize trong thực nghiệm chính:}
\begin{enumerate}
  \item \textbf{Cô lập biến cần đo}: giữ độ dài cố định $L=128$, khác
        biệt duy nhất giữa các cấu hình là nội dung biểu diễn; mọi thay
        đổi F1 quy hoàn toàn về thao tác gộp.
  \item \textbf{Compact gây nhiễu}: mỗi câu sinh $L'$ khác nhau, trộn
        lẫn ảnh hưởng của gộp với ảnh hưởng của độ dài.
  \item \textbf{Compact ở vị trí này không tăng tốc đáng kể}: 12 lớp
        backbone vẫn chạy ở $L=128$; chỉ hai lớp SA nhẹ phía sau được
        hưởng lợi -- phần nhỏ trong tổng chi phí.
\end{enumerate}

\noindent Compact và pre-BERT ToMe là hướng phát triển tăng tốc thực sự
(Chương~\ref{chap:ketluan}).

\subsection{3.2.3. Huấn luyện đa nhiệm}
\label{subsec:multitask_loss}

\begin{equation}
\mathcal{L} = w_s\,\mathcal{L}_{\text{sent}}
            + w_c\,\mathbbm{1}[\text{main\_mask}]\cdot\mathcal{L}_{\text{cat}}
\label{eq:loss_method}
\end{equation}

\texttt{main\_mask}$=$ True khi \texttt{is\_supplement=False}. Trọng
số lớp: $w_k = N/(C\cdot n_k)$. Mặc định $w_s=w_c=1.0$.

\subsection{3.4.3. Ví dụ PP1b -- LCF-CDM}
\label{subsec:ex_cdm}

CDM dùng cùng SRD nhưng cắt cứng: $m_i=1$ nếu $SRD_i\le5$, ngược lại
$m_i=0$. Vị trí 10--12 bị \textbf{xóa hoàn toàn} ($v_i\cdot0=[0,0,0,0]$).
CDM tốt hơn cho nhãn thiểu số vì tập trung ``cứng'' vào vùng khía cạnh;
CDW giữ nhiều thông tin toàn cục hơn.

\subsection{3.4.4. Ví dụ PP2a -- ToMe Bipartite}
\label{subsec:ex_bip}

Token bảo vệ: vị trí 0 (CLS), 12 (SEP), 4 (food). Pool nội bộ:
$\{1,2,3,5,6,7,8,9,10,11\}$, $nm=10$.

\noindent Chia xen kẽ (theo thứ hạng trong pool, không theo chẵn/lẻ vị trí):
\[A=\{1,3,6,8,10\},\quad B=\{2,5,7,9,11\}.\]

Ví dụ cosine giữa $v_1$=[1,1,0,1] và $v_7$=[3,2,1,2]:
\[\cos(v_1,v_7)=\frac{1\cdot3+1\cdot2+0\cdot1+1\cdot2}{\sqrt{3}\cdot\sqrt{18}}=\frac{7}{7.35}\approx0.95.\]

\begin{table}[H]
\centering
\caption{Bipartite -- chọn cặp và khử trùng}
\label{tab:ex_bip}
\small
\begin{tabular}{@{}llcl@{}}
\toprule
Token $A$ & $B$ tương đồng nhất & cos & Kết quả \\
\midrule
1 (i)        & 7 (service) & 0.95 & \cmark ghép $1\leftarrow7$ \\
3 (the)      & 2 (found)   & 0.91 & \cmark ghép $3\leftarrow2$ \\
6 (but)      & 2 (found)   & 0.85 & \xmark ($B{=}2$ đã dùng) \\

Sau bước 1: xóa vị trí 7 và 2, chuỗi còn \textbf{11 token}. Bước 2
lặp lại tương tự $\to$ $\approx$\textbf{9 token}.

\subsection{3.4.5. Ví dụ PP2b -- Sequential Local}
\label{subsec:ex_seq}

Phát hiện ranh giới: tại $j=3$: $\texttt{lcf}_3=0$ và $\texttt{lcf}_4=1$
$\Rightarrow$ vị trí 3 (\textit{the}) là \texttt{mid\_sep}, không bị gộp.

Tại token 5 (\textit{delicious}):
\begin{align*}
\cos(v_5,v_4) &= \frac{[4,3,2,2]\cdot[5,4,3,2]}{\sqrt{33}\cdot\sqrt{54}}
= \frac{42}{42.2}\approx0.995\quad\text{(trái)},\\
\cos(v_5,v_6) &= \frac{[4,3,2,2]\cdot[1,0,2,0]}{\sqrt{33}\cdot\sqrt{5}}
= \frac{8}{12.8}\approx0.625\quad\text{(phải)}.
\end{align*}

Sim trái lớn hơn $\Rightarrow$ \textit{delicious} gập vào \textbf{food}
(vị trí 4): $x_4\leftarrow\tfrac12(v_4+v_5)$;
$\texttt{lcf}_4=\max(1,0)=1$ -- \textbf{tín hiệu khía cạnh giữ nguyên}.
\textbf{Hiệu ứng cascade}: sau khi vị trí 4 cập nhật, các token kế tiếp
(but, service, was, terrible, \ldots) lần lượt gập về vị trí 4, chuỗi
rút xuống $\approx$5 token chỉ sau một lượt.

\subsection{3.4.6. Ví dụ PP2c -- Attention-Weighted}
\label{subsec:ex_attn}

Trong huấn luyện không có tín hiệu attention nên \texttt{attn\_seg}$=1$
đồng nhất. Thuật toán chọn token đầu tiên chưa loại, không bảo vệ:
vị trí 1 (\textit{i}). Cosine cao nhất với hàng xóm: vị trí 7
(\textit{service}, $\approx0.95$) $\Rightarrow$ xóa vị trí 1, cập nhật
$x_7\leftarrow\tfrac12(v_7+v_1)$. Giới hạn $\lfloor(13-2)/2\rfloor=5$
lần gộp $\Rightarrow$ chuỗi còn \textbf{8 token}; food luôn bảo vệ.

\subsection{3.4.7. Ví dụ tổ hợp: CDM + Bipartite + Resize}
\label{subsec:ex_combined}

Cấu hình \texttt{lcf\_bip\_cdm\_resize} (\texttt{use\_lcf=True},
\texttt{use\_cdm=True}, \texttt{use\_tome=True}, \texttt{resize=True},
\texttt{merge\_strategy="bipartite"}):

\begin{enumerate}
  \item \textbf{Backbone} sinh $H\in\mathbb{R}^{128\times768}$ (13 hàng
        hợp lệ, còn lại padding).
  \item \textbf{ToMe bipartite}: 13 $\to$ 9 token sau 2 bước.
        \textbf{Resize}: nội suy tuyến tính 9 $\to$ 128 hàng.
        \texttt{lcf\_vec} cũng được nội suy $\Rightarrow$ vùng khía cạnh
        ``loang'' thành dải giá trị liên tục quanh vị trí 4.
  \item \textbf{CDM}: tìm vùng $\texttt{lcf}>0.5$, suy ra center/half;
        tính mặt nạ nhị phân -- chỉ cửa sổ $[center-\alpha, center+\alpha]$
        được giữ ($m_i=1$).
  \item \textbf{Luồng cục bộ}: $H_{local}=H\odot m$.
  \item \textbf{Fusion} $\to$ SA $\to$ pooler $\to$ hai đầu phân loại
        $\to$ \textit{Positive}, \texttt{FOOD}.
\end{enumerate}

\noindent Hai điểm then chốt:
\begin{itemize}
  \item \textbf{ToMe không tạo đặc trưng mới} -- chỉ trung bình hóa các
        token tương đồng; không có tham số học.
  \item \textbf{Token \texttt{[CLS]} vào pooler} là từ $H_{fused}$ sau
        Fusion + SA, không phải \texttt{[CLS]} thô của backbone -- quyết
        định phân loại dựa trên cả hai luồng cục bộ và toàn cục.
\end{itemize}

\section*{Tóm tắt chương}
\addcontentsline{toc}{section}{Tóm tắt chương}

Chương 3 trình bày: (1) kiến trúc bốn stage của hệ thống Joint Triplet
Extraction với hai chế độ Joint/Pipeline; (2) chi tiết stage APC --
LCF (CDW/CDM, SRD, Fusion) và Token Merging (ba chiến lược, chế độ
resize/compact/pre-BERT, lý do chỉ dùng resize); (3) bộ dữ liệu, siêu
tham số và 12 cấu hình thực nghiệm; (4) năm ví dụ minh họa tường minh
với toy vectors. Kết quả thực nghiệm được trình bày ở Chương~\ref{chap:ketqua}.
