\section{Kết quả theo Sentiment và Category}
\label{sec:e2e_3label}

Phần này trình bày kết quả đánh giá trên \textbf{task 3 nhãn}:
mô hình phải dự đoán đồng thời ba thành phần
$(\textit{aspect term},\; \textit{category},\; \textit{sentiment})$
cho mỗi câu đánh giá. Một bộ ba $(\hat{a}, \hat{c}, \hat{s})$ được tính
là đúng (True Positive) \emph{khi và chỉ khi} cả ba thành phần đều khớp
với nhãn gold.

% ============================================================
\subsection{F1 theo lớp cảm xúc}
\label{subsec:sent_perclass_avg}

Bảng~\ref{tab:sent_perclass_avg} báo cáo F1 của từng nhãn cảm xúc
khi phân loại sentiment \emph{với aspect term gold}. Tập test mất cân
bằng nghiêm trọng (Positive: 83,3\%, Negative: 14,7\%, Neutral: 1,9\%),
dẫn đến F1-Neutral biến thiên lớn giữa các cấu hình.

\textbf{Model tốt nhất:}
\begin{itemize}
  \item \textbf{F1-Negative}: AWM đạt \textbf{86,36\%}
        (+5,41 điểm so với Baseline 80,95\%).
  \item \textbf{F1-Neutral}: AWM đạt \textbf{66,67\%}
        (+26,67 điểm so với Baseline 40,00\%).
  \item \textbf{F1-Positive}: LCF-CDW+SLM đạt \textbf{98,25\%}
        (Baseline = 97,90\%).
\end{itemize}

        ($\leq 1,26$ điểm).
  \item \textbf{AWM nổi bật ở nhãn thiểu số}: F1-Negative tốt nhất
        (86,36\%) và F1-Neutral tốt nhất (66,67\%). Cơ chế
        attention-weighted merging ưu tiên giữ lại token mang tín hiệu
        phân biệt cho nhãn hiếm.
  \item \textbf{LCF-CDM kết hợp AWM không cải thiện}: kết hợp LCF-CDM
        làm giảm cả ba F1 so với AWM đơn, cho thấy mặt nạ cứng CDM
        cạnh tranh với cơ chế trọng số attention.
  \item \textbf{SLM và LCF-CDW+SLM} đạt F1-Neutral = 60,00\%,
        xếp thứ hai sau AWM.
\end{itemize}

\textbf{Nhận xét:}
\begin{itemize}
  \item \textbf{SERVICE và FACILITY} (hai nhãn nhiều nhất trong train)
        ổn định ở mức 95--98\% và 87--91\% trên mọi cấu hình.
  \item \textbf{LOYALTY} đạt 92,31\% ở 10/11 cấu hình; chỉ LCF-CDM
        thấp hơn (85,71\%) do mặt nạ cứng CDM ảnh hưởng nhãn hiếm.
  \item \textbf{BRANDING và EXPERIENCE} là hai nhãn có độ biến thiên
        lớn nhất (BRAND: 72--90\%; EXP: 71--83\%), phản ánh khó khăn
        khi phân biệt các khía cạnh có ngữ nghĩa gần nhau.
  \item \textbf{BiToMe nổi bật ở EXPERIENCE} (83,08\%) và \textbf{SERVICE}
        (97,73\%), trong khi \textbf{AWM} dẫn đầu ở \textbf{FACILITY}
        (91,46\%), cho thấy mỗi chiến lược token merging phù hợp với
        phân phối nhãn khác nhau.
  \item \textbf{LCF-CDM + AWM} đạt BRANDING cao nhất (90,00\%), cho
        thấy mặt nạ cứng CDM kết hợp attention-weighted merging có lợi
        riêng cho nhãn hiếm có đặc trưng phân biệt rõ.
\end{itemize}

\textbf{Kết luận:}
\begin{itemize}
  \item \textbf{AWM là lựa chọn tốt nhất tổng thể} cho task 3 nhãn:
        đứng đầu về E2E Micro-F1, E2E Macro-F1, Oracle Acc,
        F1-Negative và F1-Neutral cùng một lúc.
  \item \textbf{BiToMe} là lựa chọn thứ hai về E2E Micro-F1 (69,32\%)
        và xuất sắc ở EXPERIENCE (+9,40 so với baseline) và SERVICE
        (+2,13), với thời gian huấn luyện thấp hơn AWM ($\approx393$\,s
        so với $719$\,s).
  \item \textbf{Kết hợp LCF + AWM không cải thiện AWM}: E2E Macro-F1
        giảm từ 57,20 (AWM) xuống 51,03 (LCF-CDM+AWM). Điều này gợi
        ý cơ chế attention-weighted merging đã tích hợp ngầm thông tin
        context và không được lợi thêm từ LCF.
\end{itemize}
